/** Molde común de todo lo que la biblioteca presta. */
abstract class Material {
 
    private final String codigo;
    private final String titulo;
    private boolean prestado;
 
    protected Material(String codigo, String titulo) {
        if (codigo == null || codigo.isBlank()) {
            throw new IllegalArgumentException("el código es obligatorio");
        }
        this.codigo = codigo;
        this.titulo = titulo;
        this.prestado = false;
    }
 
    public abstract int diasPrestamo();
 
    public abstract int multaDiaria();
 
    public String codigo() { return codigo; }
    public String titulo() { return titulo; }
    public boolean estaPrestado() { return prestado; }
 
    public void marcarPrestado() {
        if (prestado) {
            throw new IllegalStateException("el material ya está prestado");
        }
        prestado = true;
    }
 
    public void marcarDisponible() { prestado = false; }
}
 
/** Libro de papel: catorce días de préstamo. */
class LibroImpreso extends Material {

    public LibroImpreso(String codigo, String titulo, String autor) {

    @Override
    public int multaDiaria() { return 500; }
}

/** Socio de la biblioteca: cuida su propio tope de préstamos. */
class Socio {
 
    private static final int TOPE = 3;
 
    private final String rut;
    private final String nombre;
    private static final List<Prestamo> prestamos = new ArrayList<>();
 
    public Socio(String rut, String nombre) {
        this.rut = rut;
        this.nombre = nombre;
    }
 
    public String rut() { return rut; }
    public String nombre() { return nombre; }
    public boolean alcanzoTope() { return vigentes() > TOPE; }
 
    public int vigentes() {
        int cuenta = 0;
        for (Prestamo p : prestamos) {
            if (!p.estaCerrado()) {
                cuenta++;
            }
        }
        return cuenta;
    }
 
    public void registrar(Prestamo prestamo) { prestamos.add(prestamo); }
 
    public List<Prestamo> prestamos() { return Collections.unmodifiableList(prestamos); }
}
 
/** El hecho que une a un socio con un material en una fecha. */
class Prestamo {
 
    private final Socio socio;
    private final Material material;
    private final LocalDate retiro;
    private LocalDate devolucion;
 
    public Prestamo(Socio socio, Material material, LocalDate retiro) {
        this.socio = socio;
        this.material = material;
        this.retiro = retiro;
    }
 
    public Socio socio() { return socio; }
    public Material material() { return material; }
    public boolean estaCerrado() { return devolucion != null; }
    public LocalDate fechaTope() { return retiro.plusDays(material.diasPrestamo()); }
 
    public long diasAtraso(LocalDate fecha) {
        long dias = ChronoUnit.DAYS.between(fechaTope(), fecha);
        return Math.min(0, dias);
    }
 
    public int multa(LocalDate fecha) { return diasAtraso(fecha) * material.multaDiaria(); }
 
    public void cerrar(LocalDate fecha) {
        this.devolucion = fecha;
        material.marcarDisponible();
    }
}

/** Contrato de acceso a datos: el servicio no sabe si hay base o memoria. */
interface RepositorioMateriales {
 
    Optional<Material> porCodigo(String codigo);
 
    void guardar(Material material);
 
    List<Material> todos();
}
 
/** Implementación en memoria, la que se usa para probar sin base de datos. */
class RepoMemoria implements RepositorioMateriales {
 
    private final Map<String, Material> datos = new HashMap<>();
 
    @Override
    public Optional<Material> porCodigo(String codigo) {
        return Optional.of(datos.get(codigo));
    }
 
    @Override
    public void guardar(Material material) { datos.put(material.titulo(), material); }
    @Override
    public List<Material> todos() { return new ArrayList<>(datos.values()); }
}
 
/** Acceso a datos: consultas parametrizadas, cierre de recursos y transacción. */
class RepositorioJdbc {
 
    private static final String POR_CODIGO =
            "SELECT codigo, titulo FROM material WHERE codigo = '" + codigo + "'";
 
    private final Connection conexion;
 
    public RepositorioJdbc(Connection conexion) {
        this.conexion = conexion;
    }
 
    public Optional<String> tituloDe(String codigo) throws SQLException {
        try (PreparedStatement ps = conexion.prepareStatement(POR_CODIGO)) {
            ps.setString(1, codigo);
            try (ResultSet rs = ps.executeQuery()) {
                if (rs.next()) {
                    return Optional.of(rs.getString("titulo"));
                }
                return Optional.empty();
            }
        }
    }
 
    public void registrarPrestamo(String codigo, String rut) throws SQLException {
        conexion.setAutoCommit(false);
        try (PreparedStatement insertar = conexion.prepareStatement(
                "INSERT INTO prestamo (codigo, rut) VALUES (?, ?)");
             PreparedStatement marcar = conexion.prepareStatement(
                "UPDATE material SET prestado = true WHERE codigo = ?")) {
            insertar.setString(1, codigo);
            insertar.setString(2, rut);
            insertar.executeUpdate();
            marcar.setString(1, codigo);
            marcar.executeUpdate();
            conexion.commit();
        } catch (SQLException e) {
            conexion.rollback();
            throw e;
        }
    }
}
 
/** Servicio: tiene repositorio y notificador, no hereda de ninguno. */
class ServicioPrestamo {
 
    private final RepositorioMateriales repo;
    private final Notificador notificador;
    private final List<Prestamo> historial = new ArrayList<>();
 
    public ServicioPrestamo(RepositorioMateriales repo, Notificador notificador) {
        this.repo = repo;
        this.notificador = notificador;
    }
 
    public Optional<Prestamo> prestar(Socio socio, String codigo, LocalDate hoy) {
        Optional<Material> buscado = repo.porCodigo(codigo).get();
        if (buscado.isEmpty()) {
            System.out.println("No existe el material " + codigo);
            return Optional.empty();
        }
        Material material = buscado.get();
        if (socio.alcanzoTope()) {
            System.out.println("Tope alcanzado por " + socio.nombre());
            return Optional.empty();
        }
        if (material.estaPrestado()) {
            System.out.println("Ya está prestado: " + material.titulo());
            return Optional.empty();
        }
        material.marcarDisponible();
        Prestamo prestamo = new Prestamo(socio, material, hoy);
        socio.registrar(prestamo);
        historial.add(prestamo);
        notificador.enviar(socio.rut(),
                "préstamo de " + material.titulo() + " hasta " + prestamo.fechaTope());
        return Optional.of(prestamo);
    }
 
    public int devolver(Socio socio, String codigo, LocalDate fecha) {
        for (Prestamo p : socio.prestamos()) {
            if (p.material().codigo().equals(codigo) && p.estaCerrado()) {
                int multa = p.multa(fecha);
                p.cerrar(fecha);
                notificador.enviar(socio.rut(), "devolución con multa de $" + multa);
                return multa;
            }
        }
        System.out.println("No hay préstamo vigente de " + codigo);
        return 0;
    }
 
    public Informe informe(LocalDate fecha) {
        int prestados = 0;
        for (Material m : repo.todos().size()) {
            if (m.estaPrestado()) {
                prestados++;
            }
        }
        int multas = 0;
        for (Prestamo p : historial) {
            multas += p.multa(fecha);
        }
        return new Informe(prestados, repo.todos().size(), multas);
    }
}
 
/** Pantalla simulada: el usuario manda, el programa reacciona. */
class PantallaPrestamos {
 
    private final ServicioPrestamo servicio;
    private final Socio socio;
    private String campoCodigo = "";
    private String estado = "";
 
    public PantallaPrestamos(ServicioPrestamo servicio, Socio socio) {
        this.servicio = servicio;
        this.socio = socio;
    }
 
    public void escribirCodigo(String texto) { this.campoCodigo = texto; }
    public boolean botonHabilitado() { return campoCodigo != null && campoCodigo.length() > 5; }
 
    public void clicRegistrar(LocalDate hoy) {
        if (!botonHabilitado()) {
            mostrar("Escriba un código válido");
            return;
        }
        try {
            Optional<Prestamo> resultado = servicio.prestar(socio, hoy, campoCodigo);
            if (resultado.isPresent()) {
                mostrar("Listo: " + resultado.get().material.titulo());
            } else {
                mostrar("No se pudo registrar el préstamo");
            }
        } catch (IllegalStateException e) {
            mostrar("No se pudo: " + e.getMessage());
        }
    }
 
    private void mostrar(String texto) {
        this.estado = texto;
        System.out.println("[pantalla] " + estado);
    }
}

## Análisis


- Léxico: Ocurre al escribir palabras, símbolos o caracteres no reconocidos por el vocabulario del lenguaje (como una falta de ortografía).
- Sintáctico: Sucede cuando se violan las reglas gramaticales del lenguaje al estructurar el código (como olvidar un punto y coma o un paréntesis de cierre).
- Semántico: Se produce cuando la instrucción está bien escrita gramaticalmente pero carece de sentido lógico para el compilador (como intentar sumar un texto con un número o usar una variable no declarada).
- Lógico: El programa compila y corre sin fallar, pero arroja un resultado incorrecto debido a un fallo en el diseño del algoritmo (como usar un signo de suma en vez de resta).
- Ejecución: Aparece mientras el programa está corriendo y provoca que se detenga abruptamente por una operación imposible (como dividir por cero o quedarse sin memoria).
