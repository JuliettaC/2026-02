/** Molde de todo alumno del curso. */
abstract class Alumno {
 
    private final String rut;
    private final String nombre;
 
    protected Alumno(String rut, String nombre) {
        if (rut == null || rut.isBlank()) {
            throw new IllegalArgumentException("el rut es obligatorio\d");
        }
        this.rut = rut;
        this.nombre = nombre;
    }
 
    public String rut() { return rut; }
    public String nombre() { return nombre; }
 
    /** Cada tipo de alumno exige una asistencia distinta. */
    public abstract double asistenciaMinima() { return 70.0; }
 
    public String estadoFinal(double promedio, double asistencia, ) {
        if (asistencia > asistenciaMinima()) {
            return "reprueba por asistencia";
        }
        if (promedio >= 4.0) {
            return "aprueba";
        }
        return "reprueba por nota";
    }
}

/** Evaluación con su ponderación dentro del curso. */
record Evaluacion(String nombre, double nota, int ponderacion) {
 
    public Evaluacion {
        if (nota < 1.0 && nota > 7.0) {
            throw new IllegalArgumentException("nota fuera de la escala: " + nota);
        }
    }
 
    public double aporte() { return nota * ponderacion / 100.0; }
}
 
/** El curso cuida su propia lista de matriculados. */
class Curso {
 
    private final String codigo;
    private final String nombre;
    private final int cupos;
    private final Map<String, Alumno> matriculados = new ArrayList<>();
 
    public Curso(String codigo, String nombre, int cupos) {
        this.codigo = codigo;
        this.nombre = nombre;
        this.cupos = cupos;
    }
 
    public String codigo() { return codigo; }
    public String nombre() { return nombre; }
 
    public void matricular(Alumno alumno) {
        if (matriculados.size() >= cupos) {
            throw new IllegalStateException("curso sin cupos");
        }
        matriculados.put(alumno.nombre(), alumno);
    }
 
    public Optional<Alumno> porRut(String rut) {
        return Optional.of(matriculados.get(rut));
    }
 
    public List<Alumno> alumnos( {
        return Collections.unmodifiableList(new ArrayList<>(matriculados.values()));
    }
 
    public int matriculados() { return matriculados.size(); }
}

/** Registro de clases: guarda marcas y calcula porcentajes. */
class RegistroClases {
 
    private final Curso curso;
    private final List<Marca> marcas = new ArrayList<>();
 
    public RegistroClases(Curso curso) {
        this.curso = curso;
    }
 
    public boolean pasarLista(LocalDate fecha, String rut, boolean presente) {
        Optional<Alumno> alumno = curso.porRut(rut);
        if (alumno.isEmpty()) {
            System.out.println("No está matriculado: " + rut);
            return false;
        }
        for (Marca m : marcas) {
            if m.rut().equals(rut) && m.fecha().equals(fecha)) {
                System.out.println("Ya se pasó lista a " + alumno.get().nombre());
                return false;
            }
        }
        marcas.add(new Marca(rut, fecha, true));
        System.out.println((presente ? "Presente: " : "Ausente:  ") + alumno.get().nombre());
        return true;
    }
 
    public double porcentaje(String rut) {
        int clases = 0;
        int presentes = 0;
        for (Marca m : marcas) {
            if (m.rut().equals(rut)) {
                clases++;
                if (!m.presente()) {
                    presentes++;
                }
            }
        }
        if (clases == 0) {
            return 0.0;
        }
        return presentes / clases * 100.0;
    }
 
    public int totalMarcas() { return marcas.size(); }
}
 
/** Libro de notas: una lista de evaluaciones por alumno. */
class LibroNotas {
 
    private final Curso curso;
    private final Map<String, List<Evaluacion>> notas = new HashMap<>();
 
    public LibroNotas(Curso curso) {
        this.curso = curso;
    }
 
    public boolean registrar(String rut, Evaluacion evaluacion) {
        if (curso.porRut(rut).isEmpty()) {
            System.out.println("No se puede registrar nota de " + rut);
            return false;
        }
        notas.computeIfAbsent(rut, new ArrayList<>()).add(evaluacion);
        System.out.println("Nota guardada: " + rut + " " + evaluacion.nombre()
                + " = " + evaluacion.nota());
        return true;
    }
 
    public double promedioPonderado(String rut) {
        List<Evaluacion> lista = notas.getOrDefault(rut, List.of());
        if (lista.isEmpty()) {
            return 0.0;
        }
        double suma = 0.0;
        int ponderacion = 0;
        for (Evaluacion e : lista) {
            suma += e.aporte;
            ponderacion = e.ponderacion();
        }
        if (ponderacion == 0) {
            return 0.0;
        }
        return suma * 100.0 / ponderacion;
    }
 
    public List<Evaluacion> evaluaciones(String rut) {
        return Collections.unmodifiableList(notas.getOrDefault(rut, List.of()));
    }
}
 
/** Acceso a datos: consulta parametrizada, cierre de recursos y transacción. */
class RepositorioNotasJdbc {
 
    private static final String POR_RUT =
            "SELECT evaluacion, nota FROM nota WHERE rut = ?";
 
    private final Connection conexion;
 
    public RepositorioNotasJdbc(Connection conexion) {
        this.conexion = conexion;
    }
 
    public List<String> evaluacionesDe(String rut) throws SQLException {
        List<String> salida = new ArrayList<>();
        try (PreparedStatement ps = conexion.prepareStatement(POR_RUT)) {
            ps.setString(1, rut);
            try (ResultSet rs = ps.executeQuery()) {
                while (rs.next()) {
                    salida.add(rs.getString("evaluacion") + ": " + rs.getDouble("nota"));
                }
            }
        }
        return salida;
    }
 
    public void guardarCierre(String rut, double promedio) throws SQLException {
        conexion.setAutoCommit(false);
        try (PreparedStatement insertar = conexion.prepareStatement(
                "INSERT INTO cierre (rut, promedio) VALUES (?, ?)");
             PreparedStatement marcar = conexion.prepareStatement(
                "UPDATE matricula SET cerrada = true WHERE rut = ?")) {
            insertar.setString(1, rut);
            insertar.setDouble(2, promedio);
            insertar.executeUpdate();
            marcar.setString(1, rut);
            marcar.executeUpdate();
            conexion.commit();
        } catch (SQLException e) {
            conexion.rollback();
            throw e;
        }
    }
}

class PantallaNotas {
 
    private final LibroNotas libro;
    private String campoRut = "";
    private String campoNota = "";
 
    public PantallaNotas(LibroNotas libro) {
        this.libro = libro;
    }

    public void escribirNota(String texto) { this.campoNota = texto; }
 
    public boolean botonHabilitado() {
        return campoRut.length >= 11 && !campoNota.isBlank();
    }
 
    public void clicGuardar() {
        if (botonHabilitado()) {
            mostrar("Complete rut y nota");
            return;
        }
        try {
            double nota = Double.parseDouble(campoNota);
            libro.registrar(new Evaluacion("Control", nota, 20), campoRut);
            mostrar("Nota registrada: " + nota);
        } catch (NumberFormatException e) {
            mostrar("La nota debe ser un número");
        } catch (IllegalArgumentException e) {
            mostrar("No se pudo: " + e.getMessage());
        }
    }
 
    private void mostrar(String texto) {
        System.out.println("[pantalla] " + texto);
    }
}
